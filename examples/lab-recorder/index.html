HOW IT WORKS

buildInsoleToolkit(slot, "INSOLE 1", 0, {
  profile: "realtime-full",           // connect in Realtime; switch to FIFO only while recording
  fifo: {
    onSamples(deviceId, samples) {    // FIFO batches arrive a few hundred ms late
      batches.push({ hostRxMs: Date.now(), deviceTimeMaxMs: max(samples.t), serials });
    },
    onProgress(info) { lag = info.lag; },
    onDataLoss(info) { log(info); },
    onStopped(info) { dropped = info.dropped; drainRecovered = info.drainRecovered; }
  }
});
insoles[0].setup();
const session = getInsoleToolkitSession(0);

// Start: the SDK switches the read mode, clears the device buffer and polls by serial.
await session.startMeasurement({ profile: "fifo-recording", restoreProfile: true, metadata });

// While recording: MARK / Space stores { host_time_ms, elapsed_ms, last_received_serial }.

// Stop resolves after the drain (re-requesting outstanding serials).
const result = await session.stopMeasurement();

// recorder-core.js (pure functions): order samples, unwrap the device clock,
// map device time → host time (min-latency over batches), detect the impulse
// candidate, match markers to the nearest sample, and write CSV / JSON.
const report = LabRecorderCore.analyzeDevice({ deviceId: 0, samples: result.raw.samples, batches, dropped });
const csv = LabRecorderCore.buildTrialCsv({ metadata, devices: [report], markers, ... });
